📚 Arrays

const arr = [1, 2, 3, 4, 5];

// Acesso e info
arr[0]                    // 1
arr.length                // 5
arr.at(-1)                // 5 (último)

// Adicionar/Remover
arr.push(6)               // adiciona no fim → [1,2,3,4,5,6]
arr.pop()                 // remove do fim → [1,2,3,4,5]
arr.unshift(0)            // adiciona no início → [0,1,2,3,4,5]
arr.shift()               // remove do início → [1,2,3,4,5]
arr.splice(2, 1)          // remove 1 item no índice 2
arr.splice(2, 0, 'new')   // insere 'new' no índice 2

// Busca
arr.indexOf(3)            // 2
arr.includes(3)           // true
arr.find(x => x > 3)      // 4 (primeiro match)
arr.findIndex(x => x > 3) // 3 (índice do primeiro match)

// Transformação (retornam NOVO array)
arr.map(x => x * 2)           // [2, 4, 6, 8, 10]
arr.filter(x => x > 2)        // [3, 4, 5]
arr.reduce((acc, x) => acc + x, 0) // 15

// Iteração
arr.forEach(x => console.log(x))
arr.every(x => x > 0)         // true (todos passam)
arr.some(x => x > 4)          // true (algum passa)

// Ordenação
arr.sort((a, b) => a - b)     // crescente
arr.sort((a, b) => b - a)     // decrescente
arr.reverse()                 // inverte

// Outros
arr.join(", ")                // "1, 2, 3, 4, 5"
arr.concat([6, 7])            // [1, 2, 3, 4, 5, 6, 7]
arr.slice(1, 3)               // [2, 3]
arr.flat()                    // achata arrays aninhados
[...arr]                      // cópia do array
[...new Set(arr)]             // remove duplicados

Git Cheat Sheet

Comandos essenciais para controle de versão.

🚀 Início

# Criar repositório
git init

# Clonar repositório
git clone <url>
git clone <url> nome-pasta

# Status
git status
git status -s    # resumido

# Adicionar arquivos
git add arquivo.js
git add .        # todos
git add -A       # todos (incluindo deletados)
git add *.js     # padrão

# Commit
git commit -m "mensagem"
git commit -am "mensagem"  # add + commit

# Ver histórico
git log
git log --oneline
git log --oneline --graph --all
git log -p       # com diff
git log -n 5     # últimos 5

🌿 Branches

# Listar branches
git branch       # locais
git branch -r    # remotas
git branch -a    # todas

# Criar branch
git branch nome-branch
git checkout -b nome-branch  # criar e mudar
git switch -c nome-branch    # novo modo

# Mudar de branch
git checkout nome-branch
git switch nome-branch

# Renomear
git branch -m novo-nome
git branch -m antigo novo

# Deletar
git branch -d nome    # seguro
git branch -D nome    # forçado

# Merge
git merge nome-branch
git merge --no-ff nome  # sempre criar commit

☁️ Remoto

# Ver remotos
git remote -v

# Adicionar remoto
git remote add origin <url>

# Mudar URL
git remote set-url origin <nova-url>

# Push (enviar)
git push origin main
git push -u origin main  # setar upstream
git push --all           # todas branches
git push --force-with-lease  # forçado seguro

# Pull (baixar e merge)
git pull origin main
git pull --rebase    # rebase ao invés de merge

# Fetch (só baixar)
git fetch origin
git fetch --all

# Deletar branch remota
git push origin --delete nome-branch

↩️ Desfazer

# Descartar mudanças locais
git checkout -- arquivo.js
git restore arquivo.js

# Unstage (remove do staging)
git reset HEAD arquivo.js
git restore --staged arquivo.js

# Desfazer último commit
git reset --soft HEAD~1  # mantém staged
git reset HEAD~1         # mantém unstaged
git reset --hard HEAD~1  # PERDE tudo

# Editar último commit
git commit --amend -m "nova mensagem"
git commit --amend --no-edit  # adicionar arquivo

# Reverter commit (cria novo)
git revert <commit-hash>

# Recuperar branch deletada
git reflog
git checkout -b nome <hash>

📦 Stash

# Guardar mudanças
git stash
git stash save "descrição"
git stash -u   # incluir untracked

# Listar stashes
git stash list

# Aplicar stash
git stash pop      # aplica e remove
git stash apply    # aplica e mantém
git stash pop stash@{2}  # específico

# Ver conteúdo
git stash show
git stash show -p  # com diff

# Deletar
git stash drop stash@{0}
git stash clear    # todos

🔍 Diff & Compare

# Ver diferenças
git diff                 # working vs staging
git diff --staged        # staging vs último commit
git diff HEAD            # working vs último commit
git diff commit1 commit2 # entre commits
git diff branch1 branch2 # entre branches

# Mostrar arquivo específico
git diff arquivo.js

# Ver último commit
git show
git show <commit-hash>

# Blame (quem alterou)
git blame arquivo.js
git blame -L 10,20 arquivo.js  # linhas 10-20

# Buscar
git log --grep="bugfix"
git log -S "funcaoX"     # busca no código

🏷️ Tags

# Listar tags
git tag
git tag -l "v1.*"

# Criar tag
git tag v1.0.0           # leve
git tag -a v1.0.0 -m "Release 1.0"  # anotada

# Tag em commit específico
git tag v1.0.0 <commit-hash>

# Push tags
git push origin v1.0.0
git push origin --tags

# Deletar tag
git tag -d v1.0.0        # local
git push origin :refs/tags/v1.0.0  # remota

# Checkout tag
git checkout v1.0.0

🔄 Rebase

# Rebase simples
git checkout feature
git rebase main

# Rebase interativo
git rebase -i HEAD~3

# Comandos no rebase interativo:
# pick   = usar commit
# reword = editar mensagem
# edit   = pausar para editar
# squash = juntar com anterior
# fixup  = squash sem mensagem
# drop   = remover commit

# Continuar após resolver conflito
git rebase --continue

# Abortar rebase
git rebase --abort

# Squash: juntar últimos N commits
git rebase -i HEAD~N
# marque todos exceto primeiro como "squash"
